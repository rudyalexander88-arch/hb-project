function instalarModuloAsistenciaPersonal() {
  const ss = SpreadsheetApp.openById(CODIGO_SHEET_ID);
  Object.keys(AP_ENCABEZADOS).forEach(function(nombre) {
    let hoja = ss.getSheetByName(nombre);
    if (!hoja) hoja = ss.insertSheet(nombre);
    const encabezados = AP_ENCABEZADOS[nombre];
    if (hoja.getLastRow() === 0) {
      hoja.getRange(1,1,1,encabezados.length).setValues([encabezados]);
      hoja.setFrozenRows(1);
      hoja.getRange(1,1,1,encabezados.length).setFontWeight("bold").setBackground("#e9eef3");
    } else {
      const actuales = hoja.getRange(1,1,1,Math.max(1,hoja.getLastColumn())).getValues()[0].map(String);
      const faltantes = encabezados.filter(function(e){ return actuales.indexOf(e) < 0; });
      if (faltantes.length) hoja.getRange(1,actuales.length+1,1,faltantes.length).setValues([faltantes]);
    }
  });
  if (typeof instalarModuloAvisosSistema === "function") {
    instalarModuloAvisosSistema();
  }
  crearTemporizadoresAsistenciaPersonal();
  return {ok:true,mensaje:"Módulo de asistencia instalado correctamente."};
}

function obtenerResumenAsistenciaPersonal(data) {
  try {
    const acceso = apValidarResponsable_(data);
    const fecha = apFecha_(data && data.fecha) || new Date();
    const turno = String(data && data.turno || "4-3 A").trim();
    const claveFecha = apFechaTexto_(fecha);
    const usuarios = apUsuariosActivos_(turno,fecha);
    const asistencia = apLeer_(AP_HOJAS.ASISTENCIA).filter(function(r) {
      return apFechaTexto_(apFecha_(r.Fecha)) === claveFecha && apClave_(r.Turno) === apClave_(turno);
    });
    const porEmpleado = {};
    asistencia.forEach(function(r) { porEmpleado[String(r.Empleado_ID)] = r; });
    return respuesta({ok:true,data:{
      fecha:claveFecha,
      turno:turno,
      colaboradores:usuarios.map(function(u) {
        return {idEmpleado:u.id,nombre:u.nombre,rol:u.rol,usuario:u.usuario,registro:porEmpleado[u.id] || null};
      }),
      coberturas:asistencia.filter(r=>apClave_(r.Es_Cobertura)==="SI").map(apSerializarRegistro_),
      disponiblesCobertura:apOrdenarUsuariosTurnoNombre_(Object.values(apUsuariosMapa_()).filter(function(u){
        return ["ACTIVO","ACTIVA"].includes(apClave_(u.estado)) && apEsColaboradorAsistencia_(u) &&
          apClave_(u.turno)!==apClave_(turno) && !apNoDisponibleEnFecha_(u.id,fecha);
      })).map(function(u){return {idEmpleado:u.id,nombre:u.nombre,rol:u.rol,turno:u.turno};}),
      noRequeridos:apNovedadesProgramadasTurno_(turno,fecha),
      lineasTrabajo:apListaConfig_("Linea_trabajo"),
      resumen:{registrados:asistencia.length,asistieron:asistencia.filter(r=>["ASISTIO","TARDANZA"].includes(apClave_(r.Estado_Asistencia))).length,
        faltaron:asistencia.filter(r=>apClave_(r.Estado_Asistencia)==="FALTO").length,
        vacaciones:apNovedadesProgramadasTurno_(turno,fecha).filter(x=>x.tipo==="VACACIONES").length,
        bonoFelicidad:apNovedadesProgramadasTurno_(turno,fecha).filter(x=>x.tipo==="BONO FELICIDAD").length},
      responsable:acceso
    }});
  } catch (error) { return respuesta({ok:false,mensaje:apError_(error)}); }
}

function guardarAsistenciaPersonal(data) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    const responsable = apValidarResponsable_(data);
    const fecha = apFecha_(data.fecha);
    const turno = String(data.turno || "").trim();
    const registros = Array.isArray(data.registros) ? data.registros : [];
    if (!fecha || !turno || !registros.length) throw new Error("Debe indicar fecha, turno y colaboradores.");
    const usuarios = apUsuariosMapa_();
    const existentes = apLeer_(AP_HOJAS.ASISTENCIA);
    const hoja = getSheet(AP_HOJAS.ASISTENCIA);
    const encabezados = hoja.getRange(1,1,1,hoja.getLastColumn()).getValues()[0].map(String);
    let guardados = 0;

    registros.forEach(function(item) {
      const id = String(item.empleadoId || "").trim();
      const usuario = usuarios[id];
      if (!usuario) throw new Error("No existe el colaborador " + id + ".");
      const esCobertura = apClave_(item.esCobertura) === "SI";
      if (!apEsColaboradorAsistencia_(usuario) || (!esCobertura && apClave_(usuario.turno) !== apClave_(turno))) {
        throw new Error(usuario.nombre + " no pertenece al turno seleccionado o su rol no aplica para asistencia.");
      }
      const estado = apClave_(item.estadoAsistencia);
      if (["ASISTIO","FALTO","TARDANZA"].indexOf(estado) < 0) return;
      if (esCobertura && estado !== "ASISTIO") throw new Error("El personal agregado como cobertura debe figurar como asistió.");
      if (estado === "TARDANZA" && !String(item.horaLlegada || "").trim()) throw new Error("Indique la hora de llegada de " + usuario.nombre + ".");
      const tipo = estado === "FALTO" ? apClave_(item.tipoInasistencia) : "NO APLICA";
      if (estado === "FALTO" && ["INJUSTIFICADA","MEDICA","OTRA JUSTIFICADA"].indexOf(tipo) < 0) {
        throw new Error("Debe clasificar la falta de " + usuario.nombre + ".");
      }
      const detalle=String(item.detalleJustificacion||item.comentario||"").trim();
      if(estado==="FALTO"&&tipo==="OTRA JUSTIFICADA"&&!detalle)throw new Error("Debe detallar la justificación de "+usuario.nombre+".");
      const idExistente = existentes.find(function(r) {
        return String(r.Empleado_ID) === id && apFechaTexto_(apFecha_(r.Fecha)) === apFechaTexto_(fecha) && String(r.Turno) === turno;
      });
      const idAsistencia = idExistente ? String(idExistente.ID_Asistencia) : apId_("ASI");
      const descuenta = estado === "FALTO" && tipo === "INJUSTIFICADA" ? "SI" : "NO";
      let threadId = idExistente ? String(idExistente.Gmail_Thread_ID || "") : "";
      let correo = idExistente ? String(idExistente.Correo_Enviado || "NO") : "NO";

      if (estado === "FALTO" && tipo === "INJUSTIFICADA" && !threadId && apConfigSi_("Enviar_Correos_Asistencia")) {
        const envio = apEnviarCorreoFalta_(usuario, fecha, turno, tipo, item.comentario, responsable);
        threadId = envio.threadId;
        correo = "SI";
      }

      const fila = {
        ID_Asistencia:idAsistencia,Fecha:fecha,Turno:turno,Empleado_ID:id,Empleado_Nombre:usuario.nombre,
        Estado_Asistencia:estado,Tipo_Inasistencia:tipo,Detalle_Justificacion:detalle,Descuenta_Dia:descuenta,Comentario:String(item.comentario||""),
        Responsable_ID:responsable.id,Responsable_Nombre:responsable.nombre,Fecha_Registro:idExistente?idExistente.Fecha_Registro:new Date(),Fecha_Modificacion:new Date(),
        Correo_Enviado:correo,Gmail_Thread_ID:threadId,Estado_Seguimiento:estado === "FALTO" ? (tipo==="INJUSTIFICADA"?"PENDIENTE AMONESTACION":"PENDIENTE REVISION") : "CERRADO",
        Turno_Origen:usuario.turno,Es_Cobertura:esCobertura?"SI":"NO",
        Motivo_Cobertura:String(item.motivoCobertura||""),Linea_Trabajo:String(item.lineaTrabajo||""),
        Empleado_Cubierto_ID:String(item.empleadoCubiertoId||""),Empleado_Cubierto_Nombre:String(item.empleadoCubiertoNombre||""),
        Genera_Horas_Extras:"NO",ID_Hora_Extra:idExistente?String(idExistente.ID_Hora_Extra||""):"",
        Hora_Inicio_Programada:apHorarioPersonal_(turno).inicio,
        Hora_Final_Programada:apHorarioPersonal_(turno).fin,
        Hora_Entrada_Real:estado === "TARDANZA" ? String(item.horaLlegada).trim() : (idExistente ? idExistente.Hora_Entrada_Real || "" : ""),
        Horas_Trabajadas:estado === "TARDANZA" ? apHorasEntre_(String(item.horaLlegada).trim(),apHorarioPersonal_(turno).fin) : (idExistente ? idExistente.Horas_Trabajadas || "" : ""),
        Minutos_Tardanza:estado === "TARDANZA" ? apMinutosDiferencia_(apHorarioPersonal_(turno).inicio,String(item.horaLlegada).trim()) : 0,
        Origen_Marcacion_Entrada:estado === "TARDANZA" ? "MANUAL" : (idExistente ? idExistente.Origen_Marcacion_Entrada || "" : "")
      };
      apUpsert_(hoja,encabezados,"ID_Asistencia",idAsistencia,fila);

      if (estado === "FALTO" && tipo === "INJUSTIFICADA") {
        apCrearAmonestacionPendiente_(fila,usuario,responsable);
      }
      if (estado === "FALTO" && tipo !== "INJUSTIFICADA") {
        apCrearJustificacionAusencia_(fila,usuario,tipo,detalle);
      }
      if (estado === "ASISTIO" || estado === "TARDANZA") {
        const he = apGenerarHoraExtraAsistencia_(fila,usuario,responsable,item);
        if (he) {
          apActualizar_(AP_HOJAS.ASISTENCIA,"ID_Asistencia",idAsistencia,{Genera_Horas_Extras:"SI",ID_Hora_Extra:he});
        }
      }
      guardados++;
    });
    SpreadsheetApp.flush();
    return respuesta({ok:true,mensaje:"Asistencia guardada correctamente.",data:{guardados:guardados}});
  } catch (error) { return respuesta({ok:false,mensaje:apError_(error)}); }
  finally { try { lock.releaseLock(); } catch(e) {} }
}

function asignarPersonalJornadaLote(data) {
  try {
    const asignaciones = Array.isArray(data.asignaciones) ? data.asignaciones : [];
    if (!asignaciones.length) throw new Error("Seleccione al menos un colaborador para guardar el turno.");
    let guardadas = 0;
    for (let i=0;i<asignaciones.length;i++) {
      const parametros = Object.assign({}, data, asignaciones[i]);
      delete parametros.asignaciones;
      const respuestaIndividual = JSON.parse(asignarPersonalJornada(parametros).getContent());
      if (!respuestaIndividual.ok) throw new Error((guardadas ? guardadas + " asignaciones fueron guardadas. " : "") + respuestaIndividual.mensaje);
      guardadas++;
    }
    return respuesta({ok:true,mensaje:guardadas + " colaboradores asignados correctamente.",data:{guardadas:guardadas}});
  } catch (error) {
    return respuesta({ok:false,mensaje:apError_(error)});
  }
}

function asignarPersonalJornada(data){
  const bloqueo = LockService.getScriptLock();

  try {
    bloqueo.waitLock(30000);

    const responsable = apValidarResponsable_(data);
    const jornada = apBuscar_(
      AP_HOJAS.CALENDARIO,
      "ID_Jornada",
      String(data.idJornada || "")
    );
    const colaborador = apUsuarioPorId_(data.empleadoId);

    if (!jornada || !colaborador || !apEsColaboradorAsistencia_(colaborador)) {
      throw new Error("Jornada o colaborador no válido.");
    }

    const detalle = data.idDetalleJornada
      ? apBuscar_(
          AP_HOJAS.CALENDARIO_DETALLE,
          "ID_Detalle_Jornada",
          data.idDetalleJornada
        )
      : null;
    const trabajoCorrido = apClave_(data.trabajoCorrido) === "SI";

    if (data.idDetalleJornada && !detalle) {
      throw new Error("No se encontró el turno y la línea seleccionados.");
    }

    if (detalle && String(detalle.ID_Jornada) !== String(jornada.ID_Jornada)) {
      throw new Error("La línea seleccionada no corresponde a esta jornada.");
    }

    if (trabajoCorrido && !detalle) {
      throw new Error("Seleccione una línea para registrar trabajo corrido.");
    }

    let detalleSiguiente = null;

    if (trabajoCorrido) {
      const turnosOrdenados = ["A", "B", "C"].map(function(codigo) {
        return apClave_(apTurnoConfig_(codigo).codigo);
      });
      const indiceTurno = turnosOrdenados.indexOf(apClave_(detalle.Turno));

      if (indiceTurno < 0 || indiceTurno >= turnosOrdenados.length - 1) {
        throw new Error(
          "El turno " + detalle.Turno + " no tiene un turno siguiente disponible para trabajo corrido."
        );
      }

      const turnoSiguiente = turnosOrdenados[indiceTurno + 1];

      detalleSiguiente = apLeer_(AP_HOJAS.CALENDARIO_DETALLE).find(function(item) {
        return String(item.ID_Jornada) === String(jornada.ID_Jornada) &&
          apClave_(item.Turno) === turnoSiguiente &&
          apClave_(item.Linea_Trabajo) === apClave_(detalle.Linea_Trabajo);
      }) || null;

      if (!detalleSiguiente) {
        throw new Error(
          'La línea "' + detalle.Linea_Trabajo + '" no está programada en el turno siguiente ' +
          turnoSiguiente + "."
        );
      }
    }

    const asignacionesExistentes = apLeer_(AP_HOJAS.JORNADA_PERSONAL);
    const detallesAsignar = [detalle].concat(detalleSiguiente ? [detalleSiguiente] : []);

    detallesAsignar.filter(Boolean).forEach(function(detalleAsignacion) {
      const asignados = asignacionesExistentes.filter(function(item) {
        return String(item.ID_Detalle_Jornada) === String(detalleAsignacion.ID_Detalle_Jornada) &&
          apClave_(item.Estado_Asignacion) !== "CANCELADA";
      }).length;
      const requeridos = Number(detalleAsignacion.Cantidad_Requerida || 0);

      if (asignados >= requeridos) {
        throw new Error(
          'La línea "' + detalleAsignacion.Linea_Trabajo + '" del turno ' +
          detalleAsignacion.Turno + " no tiene cupos disponibles."
        );
      }

      const duplicado = asignacionesExistentes.some(function(item) {
        return String(item.ID_Jornada) === String(jornada.ID_Jornada) &&
          String(item.Empleado_ID) === String(colaborador.id) &&
          apClave_(item.Turno_Trabajado) === apClave_(detalleAsignacion.Turno) &&
          apClave_(item.Linea_Trabajo) === apClave_(detalleAsignacion.Linea_Trabajo) &&
          apClave_(item.Estado_Asignacion) !== "CANCELADA";
      });

      if (duplicado) {
        throw new Error(
          colaborador.nombre + " ya está asignado en el turno " +
          detalleAsignacion.Turno + " y la línea " + detalleAsignacion.Linea_Trabajo + "."
        );
      }
    });

    const grupo = apClave_(jornada.Grupo_Carga_Regular);
    const es43 = ["4-3 A", "4-3 B"].includes(apClave_(colaborador.turno));
    const regular = es43 && apClave_(colaborador.turno) === grupo;
    const tipo = regular
      ? "CARGA REGULAR"
      : es43
        ? "CARGA ADICIONAL"
        : "TURNO EXTRAORDINARIO";
    const genera = regular ? "NO" : "SI";
    const opciones = {
      tipoAsignacion: tipo,
      grupoCargaRegular: grupo,
      generaHorasExtras: genera,
      trabajoCorrido: trabajoCorrido ? "SI" : "NO",
      observaciones: String(data.observaciones || "")
    };

    const idAsignacion = apAgregarAsignacionJornada_(
      jornada,
      detalle,
      colaborador,
      Object.assign({}, opciones, {
        turnoTrabajado: detalle ? detalle.Turno : String(data.turnoTrabajado || ""),
        lineaTrabajo: detalle ? detalle.Linea_Trabajo : String(data.lineaTrabajo || "")
      }),
      responsable
    );

    let idAsignacionSiguiente = "";

    if (detalleSiguiente) {
      idAsignacionSiguiente = apAgregarAsignacionJornada_(
        jornada,
        detalleSiguiente,
        colaborador,
        Object.assign({}, opciones, {
          turnoTrabajado: detalleSiguiente.Turno,
          lineaTrabajo: detalleSiguiente.Linea_Trabajo
        }),
        responsable
      );
    }

    if (genera === "SI") {
      apAsegurarHoraExtraAsignacion_(idAsignacion, responsable);
      if (idAsignacionSiguiente) apAsegurarHoraExtraAsignacion_(idAsignacionSiguiente, responsable);
    }

    detallesAsignar.filter(Boolean).forEach(function(detalleAsignacion) {
      apRecontarDetalleJornada_(detalleAsignacion.ID_Detalle_Jornada);
    });

    return respuesta({
      ok: true,
      mensaje: trabajoCorrido
        ? "Colaborador asignado correctamente en ambos turnos consecutivos."
        : "Colaborador asignado correctamente.",
      data: {
        idAsignacion: idAsignacion,
        idAsignacionSiguiente: idAsignacionSiguiente,
        generaHorasExtras: genera,
        trabajoCorrido: trabajoCorrido ? "SI" : "NO"
      }
    });
  } catch (error) {
    return respuesta({ok:false,mensaje:apError_(error)});
  } finally {
    try {
      bloqueo.releaseLock();
    } catch (errorBloqueo) {}
  }
}

function revisarRecepcionesVaciasPendientes(data) {
  try {
    const encabezado = getSheet("Recepciones_Encabezado");
    const detalle = getSheet("Recepciones_Detalle");
    if (!encabezado || !detalle) throw new Error("No se encontraron las hojas de recepciones.");
    const horasMinimas = Number(apConfig_("Horas_Limpieza_Recepciones_Vacias") || 24);
    const registros = apLeer_("Recepciones_Encabezado");
    const detalles = apLeer_("Recepciones_Detalle");
    const responsables = Object.values(apUsuariosMapa_()).filter(usuario => /ENCARGADO|ADMINISTRADOR/.test(apClave_(usuario.rol)) && ["ACTIVO","ACTIVA"].includes(apClave_(usuario.estado)));
    let avisosCreados = 0;
    registros.filter(registro => {
      const creada = apFechaHoraRecepcionVacia_(registro.Fecha_Creacion);
      return ["ABIERTA","EN PROCESO","PAUSADA"].includes(apClave_(registro.Estado)) && creada && Date.now() - creada.getTime() >= horasMinimas * 3600000 && Number(registro.Total_Tarimas || 0) === 0 && Number(registro.Total_Unidades || 0) === 0 && !detalles.some(item => String(item.ID_Recepcion) === String(registro.ID_Recepcion));
    }).forEach(registro => responsables.forEach(usuario => {
      if (existeAvisoSistema_(usuario.id,"RECEPCION VACIA",registro.ID_Recepcion)) return;
      crearAvisoSistema_({empleadoId:usuario.id,empleadoNombre:usuario.nombre,rolDestino:usuario.rol,tipo:"RECEPCION VACIA",titulo:"Recepción vacía pendiente de revisión",mensaje:"La recepción " + registro.ID_Recepcion + " no tiene líneas, tarimas ni unidades y supera " + horasMinimas + " horas. Revise y apruebe su eliminación.",referenciaTipo:"RECEPCION VACIA",referenciaId:registro.ID_Recepcion,creador:{id:"SISTEMA",nombre:"Sistema Logístico PT"}});
      avisosCreados++;
    }));
    return respuesta({ok:true,mensaje:avisosCreados + " avisos de revisión generados.",data:{avisosCreados:avisosCreados}});
  } catch(error) { return respuesta({ok:false,mensaje:apError_(error)}); }
}

function aprobarEliminacionRecepcionVacia(data) {
  try {
    const responsable = apValidarEncargado_(data);
    const id = String(data.idRecepcion || "").trim();
    const registro = apBuscar_("Recepciones_Encabezado","ID_Recepcion",id);
    if (!registro || !["ABIERTA","EN PROCESO","PAUSADA"].includes(apClave_(registro.Estado)) || Number(registro.Total_Tarimas || 0) > 0 || Number(registro.Total_Unidades || 0) > 0 || apLeer_("Recepciones_Detalle").some(item => String(item.ID_Recepcion) === id)) throw new Error("La recepción ya no cumple las condiciones para ser eliminada.");
    const horasMinimas = Number(apConfig_("Horas_Limpieza_Recepciones_Vacias") || 24);
    const creada = apFechaHoraRecepcionVacia_(registro.Fecha_Creacion);
    if (!creada || Date.now() - creada.getTime() < horasMinimas * 3600000) throw new Error("La recepción todavía no alcanza el tiempo mínimo configurado.");
    const hoja = getSheet("Recepciones_Encabezado"), valores = hoja.getDataRange().getValues(), columna = valores[0].indexOf("ID_Recepcion");
    const indice = valores.findIndex((fila,numero) => numero > 0 && String(fila[columna]) === id);
    if (indice < 1) throw new Error("No se encontró la fila de recepción.");
    hoja.deleteRow(indice + 1);
    eliminarAvisosSistemaPorReferencia_("RECEPCION VACIA",id);
    return respuesta({ok:true,mensaje:"Recepción vacía eliminada con autorización de " + responsable.nombre + "."});
  } catch(error) { return respuesta({ok:false,mensaje:apError_(error)}); }
}

function apFechaHoraRecepcionVacia_(valor) {
  if (valor instanceof Date && !isNaN(valor)) return valor;
  const fecha = new Date(valor);
  return isNaN(fecha) ? apFecha_(valor) : fecha;
}

function subirDocumentoAmonestacion(data) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    const responsable = apValidarResponsable_(data);
    const id = String(data.idAmonestacion || "").trim();
    const amonestacion = apBuscar_(AP_HOJAS.AMONESTACIONES,"ID_Amonestacion",id);
    if (!amonestacion) throw new Error("No se encontró la amonestación.");
    if (["PENDIENTE DOCUMENTO","VENCIDA"].indexOf(apClave_(amonestacion.Estado)) < 0) {
      throw new Error("Esta amonestación ya fue documentada o no se encuentra disponible.");
    }
    const causa = String(data.causa === undefined ? amonestacion.Causa || "" : data.causa).trim();
    const comentario = String(data.comentario === undefined ? amonestacion.Comentario || "" : data.comentario).trim();
    if (!causa) throw new Error("Indique la causa de la amonestación.");
    amonestacion.Causa = causa;
    amonestacion.Comentario = comentario;
    const base64 = String(data.archivoBase64 || "").replace(/^data:[^;]+;base64,/,"");
    if (!base64) throw new Error("Debe seleccionar el documento PDF.");
    const blob = Utilities.newBlob(Utilities.base64Decode(base64),"application/pdf");
    const nombre = apNombreArchivoAmonestacion_(amonestacion.Empleado_Nombre,apFecha_(amonestacion.Fecha_Incidencia));
    blob.setName(nombre);
    const carpeta = apCarpetaColaborador_(amonestacion.Empleado_ID,amonestacion.Empleado_Nombre);
    const archivo = carpeta.createFile(blob);
    let threadId = String(amonestacion.Gmail_Thread_ID || "");
    if (threadId && apConfigSi_("Enviar_Correos_Asistencia")) {
      const thread = GmailApp.getThreadById(threadId);
      if (thread) thread.reply("Adjuntamos la amonestación correspondiente a " + amonestacion.Empleado_Nombre + ".\nCausa: " + causa + (comentario ? "\nComentario: " + comentario : ""),{attachments:[archivo.getBlob()]});
    } else if (apConfigSi_("Enviar_Correos_Asistencia")) {
      const para=apConfig_("Correos_GGHH_Asistencia");if(!para)throw new Error("Falta Correos_GGHH_Asistencia en Configuracion.");
      const enviado=GmailApp.createDraft(para,"Amonestación · "+amonestacion.Empleado_Nombre+" · "+apFechaTexto_(apFecha_(amonestacion.Fecha_Incidencia)),"Se adjunta el formulario firmado.",{htmlBody:"<p>Se remite la amonestación firmada de <b>"+apHtml_(amonestacion.Empleado_Nombre)+"</b>.</p><p>Causa: "+apHtml_(causa)+"</p>"+(comentario?"<p>Comentario: "+apHtml_(comentario)+"</p>":"")+"<p>Registrado por: "+apHtml_(responsable.nombre)+"</p>",attachments:[archivo.getBlob()]}).send();threadId=enviado.getThread().getId();
    }
    apActualizar_(AP_HOJAS.AMONESTACIONES,"ID_Amonestacion",id,{
      Causa:causa,Comentario:comentario,Estado:"ENVIADA RRHH",Archivo_ID:archivo.getId(),Archivo_URL:archivo.getUrl(),Fecha_Carga:new Date(),
      Fecha_Envio_RRHH:new Date(),Gmail_Thread_ID:threadId,Correo_Enviado:"SI",Fecha_Modificacion:new Date(),Registrado_Por_ID:responsable.id,Registrado_Por_Nombre:responsable.nombre
    });
    if (amonestacion.ID_Asistencia) apActualizar_(AP_HOJAS.ASISTENCIA,"ID_Asistencia",amonestacion.ID_Asistencia,{Estado_Seguimiento:"CERRADO"});
    return respuesta({ok:true,mensaje:"Documento guardado y enviado en el mismo hilo.",data:{archivoUrl:archivo.getUrl()}});
  } catch(error) { return respuesta({ok:false,mensaje:apError_(error)}); }
  finally { try { lock.releaseLock(); } catch(e) {} }
}


function procesarSeguimientoAsistenciaDiario() {
  const hoy = new Date();
  apLeer_(AP_HOJAS.AMONESTACIONES).forEach(function(r) {
    if (apClave_(r.Estado) !== "PENDIENTE DOCUMENTO") return;
    const limite = apFecha_(r.Fecha_Limite);
    if (!limite || limite > hoy) return;
    const responsable = apUsuarioPorId_(r.Registrado_Por_ID);
    if (responsable && responsable.correo) {
      MailApp.sendEmail({to:responsable.correo,subject:"Amonestación pendiente · " + r.Empleado_Nombre,
        htmlBody:"<p>Debe cargar la amonestación pendiente de <b>" + apHtml_(r.Empleado_Nombre) + "</b>.</p>"});
    }
    apActualizar_(AP_HOJAS.AMONESTACIONES,"ID_Amonestacion",r.ID_Amonestacion,{Estado:"VENCIDA"});
  });
  reactivarUsuariosFinVacaciones();
  enviarReporteQuincenalAsistenciaSiCorresponde();
}

function reactivarUsuariosFinVacaciones() {
  const hoy = new Date(); hoy.setHours(0,0,0,0);
  apLeer_(AP_HOJAS.VACACIONES).forEach(function(r) {
    const estado = apClave_(r.Estado);
    const inicio = apFecha_(r.Fecha_Inicio);
    const fin = apFecha_(r.Fecha_Final);
    if (estado === "PROGRAMADA" && inicio && inicio <= hoy && fin && fin >= hoy) {
      apCambiarEstadoUsuario_(r.Empleado_ID,"Vacaciones");
      apActualizar_(AP_HOJAS.VACACIONES,"ID_Vacacion",r.ID_Vacacion,{Estado:"ACTIVA"});
      return;
    }
    if (estado === "ACTIVA" && fin && fin < hoy) {
      apCambiarEstadoUsuario_(r.Empleado_ID,"Activo");
      apActualizar_(AP_HOJAS.VACACIONES,"ID_Vacacion",r.ID_Vacacion,{Estado:"FINALIZADA",Fecha_Reactivacion:new Date()});
    }
  });
}

function enviarReporteQuincenalAsistenciaSiCorresponde() {
  const hoy = new Date();
  const rango = apRangoQuincena_(hoy);
  if (apFechaTexto_(hoy) !== apFechaTexto_(rango.hasta)) return {ok:true,enviado:false};
  const clave = "AP_REPORTE_" + apFechaTexto_(rango.desde) + "_" + apFechaTexto_(rango.hasta);
  const propiedades = PropertiesService.getScriptProperties();
  if (propiedades.getProperty(clave) === "ENVIADO") return {ok:true,enviado:false,duplicado:true};
  const resultado = enviarReporteQuincenalAsistencia({fecha:hoy});
  if (resultado && resultado.ok) propiedades.setProperty(clave,"ENVIADO");
  return resultado;
}

function enviarReporteQuincenalAsistencia(data) {
  const fecha = apFecha_(data && data.fecha) || new Date();
  const rango = apRangoQuincena_(fecha);
  const registros = apLeer_(AP_HOJAS.ASISTENCIA).filter(function(r) {
    const f = apFecha_(r.Fecha);
    return f && f >= rango.desde && f <= rango.hasta && apClave_(r.Descuenta_Dia) === "SI";
  });
  const mapa = {};
  registros.forEach(function(r) {
    const id = String(r.Empleado_ID); if (!mapa[id]) mapa[id]={nombre:r.Empleado_Nombre,dias:0,fechas:[]};
    mapa[id].dias++; mapa[id].fechas.push(apFechaTexto_(apFecha_(r.Fecha)));
  });
  const filas = Object.keys(mapa).map(id=>"<tr><td>"+apHtml_(mapa[id].nombre)+"</td><td>"+mapa[id].dias+"</td><td>"+mapa[id].fechas.join(", ")+"</td></tr>").join("");
  const destinatarios = apConfig_("Correos_GGHH_Asistencia");
  if (!destinatarios) throw new Error("Falta Correos_GGHH_Asistencia en Configuracion.");
  MailApp.sendEmail({to:destinatarios,subject:"Reporte quincenal de asistencia · "+apFechaTexto_(rango.desde)+" a "+apFechaTexto_(rango.hasta),
    htmlBody:"<h2>Reporte quincenal de asistencia</h2><table border='1' cellpadding='6' cellspacing='0'><tr><th>Colaborador</th><th>Días a descontar</th><th>Fechas</th></tr>"+(filas||"<tr><td colspan='3'>Sin descuentos</td></tr>")+"</table>"});
  return {ok:true,enviado:true,total:registros.length};
}

function crearTemporizadoresAsistenciaPersonal() {
  ScriptApp.getProjectTriggers().filter(t=>t.getHandlerFunction()==="procesarSeguimientoAsistenciaDiario").forEach(t=>ScriptApp.deleteTrigger(t));
  ScriptApp.getProjectTriggers().filter(t=>t.getHandlerFunction()==="procesarMarcacionesSinCierre").forEach(t=>ScriptApp.deleteTrigger(t));
  const hora = Math.max(0,Math.min(23,parseInt(apConfig_("Hora_Recordatorio_Amonestacion") || "8",10)));
  ScriptApp.newTrigger("procesarSeguimientoAsistenciaDiario").timeBased().everyDays(1).atHour(hora).create();
  ScriptApp.newTrigger("procesarMarcacionesSinCierre").timeBased().everyHours(1).create();
  return {ok:true,hora:hora};
}

function registrarMarcacionPersonal(data){const lock=LockService.getScriptLock();try{lock.waitLock(30000);const u=apUsuarioMarcacion_(data),gps=apValidarUbicacion_(data),fecha=apFechaServidor_(),hora=apHoraServidor_(),tipo=apClave_(data.tipoMarcacion),accion=apClave_(data.movimiento);if(!["ENTRADA","SALIDA"].includes(accion))throw new Error("Movimiento de marcación inválido.");const sufijo=accion==="ENTRADA"?"Entrada":"Salida",gpsDatos={};gpsDatos["Hora_"+sufijo+"_Real"]=hora;gpsDatos["Origen_Marcacion_"+sufijo]="GPS";gpsDatos["Latitud_"+sufijo]=gps.latitud;gpsDatos["Longitud_"+sufijo]=gps.longitud;gpsDatos["Precision_"+sufijo+"_Metros"]=gps.precision;gpsDatos["Distancia_"+sufijo+"_Metros"]=gps.distancia;gpsDatos.Fecha_Modificacion=new Date();if(tipo==="HORAS_EXTRAS"){const asignacion=apBuscar_(AP_HOJAS.JORNADA_PERSONAL,"ID_Asignacion",data.idAsignacion);if(!asignacion||String(asignacion.Empleado_ID)!==u.id||apFechaTexto_(apFecha_(asignacion.Fecha_Trabajo))!==fecha)throw new Error("No tiene una jornada extraordinaria asignada para hoy.");if(accion==="ENTRADA"&&asignacion.Hora_Entrada_Real)throw new Error("La entrada de esta jornada ya fue registrada.");if(accion==="SALIDA"&&!asignacion.Hora_Entrada_Real)throw new Error("Primero debe registrar la entrada.");if(accion==="SALIDA"&&asignacion.Hora_Salida_Real)throw new Error("La salida de esta jornada ya fue registrada.");gpsDatos.Asistio="SI";if(accion==="SALIDA")gpsDatos.Horas_Trabajadas=apHorasEntre_(String(asignacion.Hora_Entrada_Real),hora);apActualizar_(AP_HOJAS.JORNADA_PERSONAL,"ID_Asignacion",asignacion.ID_Asignacion,gpsDatos);let idHora=String(asignacion.ID_Hora_Extra||"");if(!idHora&&apClave_(asignacion.Genera_Horas_Extras)==="SI")idHora=apAsegurarHoraExtraAsignacion_(asignacion.ID_Asignacion,u);if(idHora){const extra=Object.assign({},gpsDatos,{Estado:accion==="ENTRADA"?"EN PROCESO":"COMPLETADA"});delete extra.Asistio;apActualizar_(AP_HOJAS.HORAS_EXTRAS,"ID_Hora_Extra",idHora,extra);}}else{let asistencia=apLeer_(AP_HOJAS.ASISTENCIA).find(a=>String(a.Empleado_ID)===u.id&&apFechaTexto_(apFecha_(a.Fecha))===fecha&&apClave_(a.Turno)===apClave_(u.turno));if(accion==="SALIDA"&&(!asistencia||!asistencia.Hora_Entrada_Real))throw new Error("Primero debe registrar la entrada.");if(asistencia&&accion==="ENTRADA"&&asistencia.Hora_Entrada_Real)throw new Error("La entrada ya fue registrada.");if(asistencia&&accion==="SALIDA"&&asistencia.Hora_Salida_Real)throw new Error("La salida ya fue registrada.");const horario=apHorarioPersonal_(u.turno);if(accion==="ENTRADA"){const minutos=horario.inicio?apMinutosDiferencia_(horario.inicio,hora):0,estado=horario.inicio&&hora.slice(0,5)>String(horario.inicio).slice(0,5)?"TARDANZA":"ASISTIO",fila=Object.assign({ID_Asistencia:asistencia?asistencia.ID_Asistencia:apId_("ASI"),Fecha:apFecha_(fecha),Turno:u.turno,Empleado_ID:u.id,Empleado_Nombre:u.nombre,Turno_Origen:u.turno,Estado_Asistencia:estado,Tipo_Inasistencia:"NO APLICA",Es_Cobertura:"NO",Genera_Horas_Extras:"NO",Descuenta_Dia:"NO",Responsable_ID:u.id,Responsable_Nombre:u.nombre,Fecha_Registro:asistencia?asistencia.Fecha_Registro:new Date(),Estado_Seguimiento:"CERRADO",Hora_Inicio_Programada:horario.inicio,Hora_Final_Programada:horario.fin,Minutos_Tardanza:estado==="TARDANZA"?minutos:0},gpsDatos);if(asistencia)apActualizar_(AP_HOJAS.ASISTENCIA,"ID_Asistencia",asistencia.ID_Asistencia,fila);else apAgregar_(AP_HOJAS.ASISTENCIA,fila);}else{gpsDatos.Horas_Trabajadas=apHorasEntre_(String(asistencia.Hora_Entrada_Real),hora);apActualizar_(AP_HOJAS.ASISTENCIA,"ID_Asistencia",asistencia.ID_Asistencia,gpsDatos);}}return respuesta({ok:true,mensaje:accion==="ENTRADA"?"Entrada registrada correctamente.":"Jornada finalizada correctamente.",data:{hora:hora,distanciaMetros:gps.distancia}});}catch(error){return respuesta({ok:false,mensaje:apError_(error)});}finally{try{lock.releaseLock();}catch(e){}}}

function apPenalizacionSemanal_(id,fecha){const dia=apFecha_(fecha),numero=dia.getDay()||7,inicio=apSumarDias_(dia,1-numero),fin=apSumarDias_(inicio,7),previas=[AP_HOJAS.ASISTENCIA,AP_HOJAS.JORNADA_PERSONAL].reduce((total,nombre)=>total+apLeer_(nombre).filter(r=>String(r.Empleado_ID)===String(id)&&apClave_(r.Incidencia_Sin_Cierre)==="SI"&&apFecha_(r.Fecha||r.Fecha_Trabajo)>=inicio&&apFecha_(r.Fecha||r.Fecha_Trabajo)<fin).length,0),indice=Math.min(3,previas+1);return Number(apConfiguracionObligatoria_("Penalizacion_Sin_Cierre_"+indice+"_Minutos"));}
function procesarMarcacionesSinCierre(){try{const zona=apZonaHoraria_(),ahora=new Date(),fechaHoy=Utilities.formatDate(ahora,zona,"yyyy-MM-dd"),horaAhora=Utilities.formatDate(ahora,zona,"HH:mm"),gracia=Number(apConfiguracionObligatoria_("Minutos_Gracia_Cierre_Jornada"));let procesadas=0;[{nombre:AP_HOJAS.ASISTENCIA,id:"ID_Asistencia",fecha:"Fecha"},{nombre:AP_HOJAS.JORNADA_PERSONAL,id:"ID_Asignacion",fecha:"Fecha_Trabajo"}].forEach(tipo=>{apLeer_(tipo.nombre).filter(r=>r.Hora_Entrada_Real&&!r.Hora_Salida_Real&&apClave_(r.Incidencia_Sin_Cierre)!=="SI").forEach(r=>{const fecha=apFechaTexto_(apFecha_(r[tipo.fecha])),horario=apHorarioPersonal_(r.Turno||r.Turno_Trabajado),fin=String(r.Hora_Final_Programada||horario.fin||"");if(!fin)return;const vencida=fecha<fechaHoy||(fecha===fechaHoy&&apMinutosDiferencia_(fin,horaAhora)>=gracia&&horaAhora>=fin.slice(0,5));if(!vencida)return;const penalizacion=apPenalizacionSemanal_(r.Empleado_ID,fecha),horas=Math.max(0,Number(apHorasEntre_(String(r.Hora_Entrada_Real),fin))-penalizacion/60),cambios={Hora_Salida_Real:fin,Horas_Trabajadas:Math.round(horas*100)/100,Minutos_Penalizacion:penalizacion,Incidencia_Sin_Cierre:"SI",Origen_Marcacion_Salida:"CIERRE AUTOMATICO",Fecha_Cierre_Automatico:ahora,Fecha_Modificacion:ahora};apActualizar_(tipo.nombre,tipo.id,r[tipo.id],cambios);if(tipo.nombre===AP_HOJAS.JORNADA_PERSONAL&&r.ID_Hora_Extra)apActualizar_(AP_HOJAS.HORAS_EXTRAS,"ID_Hora_Extra",r.ID_Hora_Extra,Object.assign({Estado:"COMPLETADA CON INCIDENCIA"},cambios));procesadas++;});});return{ok:true,procesadas:procesadas};}catch(error){console.error("Error procesando marcaciones sin cierre: "+apError_(error));return{ok:false,mensaje:apError_(error)};}}
