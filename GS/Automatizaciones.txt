/**
 * Elimina únicamente los triggers de reportes de despacho.
 *
 * @return {Object}
 */
function eliminarTemporizadoresReportesDespacho() {

  const triggers =
    ScriptApp.getProjectTriggers();

  let eliminados = 0;

  triggers.forEach(function(trigger) {

    const funcion =
      trigger.getHandlerFunction();

    if (
      FUNCIONES_TEMPORIZADORES_DESPACHO_
        .indexOf(funcion) !== -1
    ) {

      ScriptApp.deleteTrigger(
        trigger
      );

      eliminados++;
    }
  });

  return {
    exito: true,
    eliminados:
      eliminados
  };
}


/**
 * Lista los temporizadores actuales del módulo.
 *
 * @return {Object}
 */
function listarTemporizadoresReportesDespacho() {

  const triggers =
    ScriptApp.getProjectTriggers();

  const lista =
    triggers
      .filter(function(trigger) {

        return (
          FUNCIONES_TEMPORIZADORES_DESPACHO_
            .indexOf(
              trigger.getHandlerFunction()
            ) !== -1
        );
      })
      .map(function(trigger) {

        return {
          funcion:
            trigger.getHandlerFunction(),

          evento:
            String(
              trigger.getEventType()
            ),

          idUnico:
            trigger.getUniqueId()
        };
      });

  const resultado = {
    exito: true,
    total:
      lista.length,
    temporizadores:
      lista
  };

  console.log(
    JSON.stringify(
      resultado,
      null,
      2
    )
  );

  return resultado;
}

/**
 * Crea un trigger diario.
 *
 * @param {string} nombreFuncion
 * @param {Object} hora
 * @param {string} zonaHoraria
 * @return {Object}
 */
function crearTemporizadorDiario_(
  nombreFuncion,
  hora,
  zonaHoraria
) {

  const trigger =
    ScriptApp
      .newTrigger(
        nombreFuncion
      )
      .timeBased()
      .atHour(
        hora.hora
      )
      .nearMinute(
        hora.minuto
      )
      .everyDays(1)
      .inTimezone(
        zonaHoraria
      )
      .create();

  return {
    funcion:
      nombreFuncion,
    hora:
      String(hora.hora)
        .padStart(2, '0') +
      ':' +
      String(hora.minuto)
        .padStart(2, '0'),
    zonaHoraria:
      zonaHoraria,
    idUnico:
      trigger.getUniqueId()
  };
}


/**
 * Lee y convierte una hora de Configuracion.
 *
 * Acepta:
 * - Hora real de Google Sheets.
 * - 17:00
 * - 5:00 PM
 *
 * @param {string} clave
 * @param {string} valorPredeterminado
 * @return {Object}
 */
function obtenerHoraAutomatizacion_(
  clave,
  valorPredeterminado
) {

  const valor =
    obtenerConfiguracionAutomatizaciones_(
      clave,
      valorPredeterminado
    );

  if (
    Object.prototype.toString.call(valor) ===
      '[object Date]' &&
    !isNaN(valor.getTime())
  ) {

    const textoHora =
      Utilities.formatDate(
        valor,
        obtenerZonaHorariaAutomatizaciones_(),
        'HH:mm'
      );

    return convertirTextoHoraAutomatizacion_(
      textoHora
    );
  }

  return convertirTextoHoraAutomatizacion_(
    String(valor || valorPredeterminado)
      .trim()
  );
}


/**
 * Convierte texto de hora.
 *
 * @param {string} texto
 * @return {Object}
 */
function convertirTextoHoraAutomatizacion_(
  texto
) {

  let coincidencia =
    String(texto || '')
      .trim()
      .match(
        /^([01]?\d|2[0-3]):([0-5]\d)$/
      );

  if (coincidencia) {

    return {
      hora:
        Number(coincidencia[1]),
      minuto:
        Number(coincidencia[2])
    };
  }

  coincidencia =
    String(texto || '')
      .trim()
      .match(
        /^(\d{1,2}):([0-5]\d)\s*(AM|PM)$/i
      );

  if (coincidencia) {

    let hora =
      Number(coincidencia[1]);

    const minuto =
      Number(coincidencia[2]);

    const periodo =
      coincidencia[3]
        .toUpperCase();

    if (
      periodo === 'PM' &&
      hora < 12
    ) {
      hora += 12;
    }

    if (
      periodo === 'AM' &&
      hora === 12
    ) {
      hora = 0;
    }

    return {
      hora:
        hora,
      minuto:
        minuto
    };
  }

  throw new Error(
    'La hora "' +
    texto +
    '" no tiene un formato válido.'
  );
}


/**
 * Obtiene la zona horaria.
 *
 * @return {string}
 */
function obtenerZonaHorariaAutomatizaciones_() {

  const configurada =
    obtenerConfiguracionAutomatizaciones_(
      'Zona_Horaria',
      ''
    );

  return (
    String(configurada || '').trim() ||
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSpreadsheetTimeZone() ||
    Session.getScriptTimeZone() ||
    'America/Santo_Domingo'
  );
}


/**
 * Lee una configuración.
 *
 * @param {string} claveBuscada
 * @param {*} valorPredeterminado
 * @return {*}
 */
function obtenerConfiguracionAutomatizaciones_(
  claveBuscada,
  valorPredeterminado
) {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();

  const hoja =
    ss.getSheetByName('Configuracion');

  if (!hoja) {
    throw new Error(
      'No se encontró la hoja "Configuracion".'
    );
  }

  const ultimaFila =
    hoja.getLastRow();

  if (ultimaFila < 1) {
    return valorPredeterminado;
  }

  const datos =
    hoja
      .getRange(
        1,
        1,
        ultimaFila,
        Math.max(2, hoja.getLastColumn())
      )
      .getValues();

  const claveObjetivo =
    normalizarClaveAutomatizaciones_(
      claveBuscada
    );

  for (let i = 0; i < datos.length; i++) {

    const claveFila =
      normalizarClaveAutomatizaciones_(
        datos[i][0]
      );

    if (claveFila === claveObjetivo) {
      return datos[i][1];
    }
  }

  return valorPredeterminado;
}


/**
 * Normaliza claves.
 *
 * @param {*} valor
 * @return {string}
 */
function normalizarClaveAutomatizaciones_(
  valor
) {

  return String(valor || '')
    .trim()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/\s+/g, '_')
    .toUpperCase();
}
