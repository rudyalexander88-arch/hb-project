function listarChoferes() {

  const hoja =
    getSheet(SHEETS.CHOFERES);

  if (!hoja) {

    return respuesta({
      ok: false,
      mensaje: "No existe la hoja Choferes."
    });

  }

  const datos =
    hoja
      .getDataRange()
      .getDisplayValues();

  if (datos.length <= 1) {

    return respuesta({
      ok: true,
      data: []
    });

  }

  const encabezados = datos[0].map(function(valor) {
    return String(valor || "").trim();
  });

  const indiceId =
    encabezados.indexOf("ID_Chofer");

  const indiceNombre =
    encabezados.indexOf("Nombre_Chofer");

  const indiceCedula =
    encabezados.indexOf("Cedula");

  const indiceContenedor =
    encabezados.indexOf("Contenedor_porDefecto");

  const indiceChasis =
    encabezados.indexOf("Chasis_porDefecto");

  const indiceEstado =
    encabezados.indexOf("Estado");

  if (
    indiceId === -1 ||
    indiceNombre === -1 ||
    indiceEstado === -1
  ) {

    return respuesta({
      ok: false,
      mensaje:
        "Faltan columnas requeridas en la hoja Choferes."
    });

  }

  const lista = datos
    .slice(1)
    .map(function(fila) {

      const id =
        String(
          fila[indiceId] || ""
        ).trim();

      const nombre =
        String(
          fila[indiceNombre] || ""
        ).trim();

      const cedula =
        indiceCedula !== -1
          ? String(
              fila[indiceCedula] || ""
            ).trim()
          : "";

      const contenedorPorDefecto =
        indiceContenedor !== -1
          ? String(
              fila[indiceContenedor] || ""
            ).trim()
          : "";

      const chasisPorDefecto =
        indiceChasis !== -1
          ? String(
              fila[indiceChasis] || ""
            ).trim()
          : "";

      const estado =
        String(
          fila[indiceEstado] || ""
        ).trim();

      return {
        id: id,
        nombre: nombre,
        cedula: cedula,
        contenedorPorDefecto: contenedorPorDefecto,
        chasisPorDefecto: chasisPorDefecto,
        estado: estado,
        texto: id + " - " + nombre
      };

    })
    .filter(function(item) {

      return (
        item.id !== "" &&
        item.nombre !== "" &&
        item.estado.toLowerCase() ===
          "activo"
      );

    });

  return respuesta({
    ok: true,
    data: lista
  });

}

function listarPersonalCalidad() {

  const hoja = getSheet(SHEETS.CALIDAD);

  if (!hoja) {
    return respuesta({
      ok: false,
      mensaje: "No existe la hoja Calidad."
    });
  }

  const datos = hoja.getDataRange().getDisplayValues();

  if (datos.length <= 1) {
    return respuesta({
      ok: true,
      data: []
    });
  }

  datos.shift();

  const lista = datos
    .map(function(fila) {

      return {
        id: String(fila[0] || "").trim(),
        nombre: String(fila[1] || "").trim(),
        estado: String(fila[2] || "").trim(),
        texto:
          String(fila[0] || "").trim() +
          " - " +
          String(fila[1] || "").trim()
      };

    })
    .filter(function(item) {
      return item.estado.toLowerCase() === "activo";
    });

  return respuesta({
    ok: true,
    data: lista
  });

}

        indiceTipo !== -1
          ? String(
              fila[indiceTipo] || ""
            ).trim()
          : "";

      const identificacion =
        String(
          fila[indiceIdentificacion] || ""
        ).trim();

      const tipoContenedor =
        indiceTipoContenedor !== -1
          ? String(
              fila[indiceTipoContenedor] || ""
            ).trim()
          : "";

      const tamanoContenedor =
        indiceTamanoContenedor !== -1
          ? String(
              fila[indiceTamanoContenedor] || ""
            ).trim()
          : "";

      const estado =
        String(
          fila[indiceEstado] || ""
        ).trim();

      return {
        id: id,
        tipo: tipo,
        identificacion: identificacion,
        tipoContenedor: tipoContenedor,
        tamanoContenedor: tamanoContenedor,
        estado: estado,
        texto:
          id + " - " + identificacion
      };

    })
    .filter(function(item) {

      return (
        item.id !== "" &&
        item.identificacion !== "" &&
        item.estado.toLowerCase() ===
          "activo"
      );

    });

  return respuesta({
    ok: true,
    data: lista
  });

}


/**
 * Guarda un contenedor nuevo en Unidades_Carga.
 * Opcionalmente lo establece como predeterminado
 * para el chofer seleccionado.
 */
function guardarUnidadCargaNueva(data) {

  const bloqueo =
    LockService.getScriptLock();

  bloqueo.waitLock(30000);

  try {

    const identificacion =
      String(
        data.identificacion || ""
      ).trim();

    const tipoContenedor =
      String(
        data.tipoContenedor || ""
      ).trim();

    const tamanoContenedor =
      String(
        data.tamanoContenedor || ""
      ).trim();

    const idChofer =
      String(
        data.idChofer || ""
      ).trim();

    const establecerPredeterminado =
      data.establecerPredeterminado === true;

    if (!identificacion) {

      return respuesta({
        ok: false,
        mensaje:
          "Debe indicar la identificación del contenedor."
      });

    }

    if (
      establecerPredeterminado &&
      !idChofer
    ) {

      return respuesta({
        ok: false,
        mensaje:
          "Debe seleccionar un chofer para establecer el contenedor predeterminado."
      });

    }

    const hojaUnidades =
      getSheet(SHEETS.UNIDADES_CARGA);

    const hojaChoferes =
      getSheet(SHEETS.CHOFERES);

    if (!hojaUnidades || !hojaChoferes) {

      return respuesta({
        ok: false,
        mensaje:
          "No se encontraron las hojas Unidades_Carga o Choferes."
      });

    }

    const datosUnidades =
      hojaUnidades
        .getDataRange()
        .getDisplayValues();

    const encabezadosUnidades =
      datosUnidades[0].map(function(valor) {
        return String(valor || "").trim();
      });

    const indiceId =
      encabezadosUnidades.indexOf(
        "ID_Unidad"
      );

    const indiceTipo =
      encabezadosUnidades.indexOf(
        "Tipo"
      );

    const indiceIdentificacion =
      encabezadosUnidades.indexOf(
        "Identificacion"
      );

    const indiceTipoContenedor =
      encabezadosUnidades.indexOf(
        "Tipo_Contenedor"
      );

    const indiceTamano =
      encabezadosUnidades.indexOf(
        "Tamano_Contenedor"
      );

    const indiceEstado =
      encabezadosUnidades.indexOf(
        "Estado"
      );

    if (
      indiceId === -1 ||
      indiceTipo === -1 ||
      indiceIdentificacion === -1 ||
      indiceEstado === -1
    ) {

      return respuesta({
        ok: false,
        mensaje:
          "Faltan columnas requeridas en Unidades_Carga."
      });

    }

    const identificacionNormalizada =
      identificacion.toLowerCase();

    let idUnidad = "";
    let filaUnidad = -1;

    for (
      let i = 1;
      i < datosUnidades.length;
      i++
    ) {

      const identificacionExistente =
        String(
          datosUnidades[i][
            indiceIdentificacion
          ] || ""
        )
          .trim()
          .toLowerCase();

      if (
        identificacionExistente ===
        identificacionNormalizada
      ) {

        idUnidad =
          String(
            datosUnidades[i][indiceId] || ""
          ).trim();

        filaUnidad = i + 1;
        break;

      }

    }

    if (filaUnidad === -1) {

      idUnidad =
        generarSiguienteIdUnidad(
          datosUnidades,
          indiceId
        );

      const filaNueva =
        encabezadosUnidades.map(
          function(encabezado) {

            const valores = {
              ID_Unidad: idUnidad,
              Tipo: "Contenedor",
              Identificacion:
                identificacion,
              Tipo_Contenedor:
                tipoContenedor,
              Tamano_Contenedor:
                tamanoContenedor,
              Estado: "Activo"
            };

            return valores[encabezado] !==
              undefined
                ? valores[encabezado]
                : "";

          }
        );

      hojaUnidades.appendRow(filaNueva);

    } else {

      hojaUnidades
        .getRange(
          filaUnidad,
          indiceEstado + 1
        )
        .setValue("Activo");

      if (
        indiceTipoContenedor !== -1 &&
        tipoContenedor
      ) {

        hojaUnidades
          .getRange(
            filaUnidad,
            indiceTipoContenedor + 1
          )
          .setValue(tipoContenedor);

      }

      if (
        indiceTamano !== -1 &&
        tamanoContenedor
      ) {

        hojaUnidades
          .getRange(
            filaUnidad,
            indiceTamano + 1
          )
          .setValue(tamanoContenedor);

      }

    }

    if (establecerPredeterminado) {

      actualizarContenedorPredeterminadoChofer(
        hojaChoferes,
        idChofer,
        idUnidad
      );

    }

    SpreadsheetApp.flush();

    return respuesta({
      ok: true,
      mensaje:
        filaUnidad === -1
          ? "Contenedor guardado correctamente."
          : "El contenedor ya existía y fue reutilizado.",
      data: {
        unidad: {
          id: idUnidad,
          tipo: "Contenedor",
          identificacion:
            identificacion,
          tipoContenedor:
            tipoContenedor,
          tamanoContenedor:
            tamanoContenedor,
          estado: "Activo",
          texto:
            idUnidad +
            " - " +
            identificacion
        },
        establecidoComoPredeterminado:
          establecerPredeterminado
      }
    });

  } finally {

    bloqueo.releaseLock();

  }

}


function generarSiguienteIdUnidad(
  datos,
  indiceId
) {

  let prefijo = "UC";
  let separador = "";
  let longitudNumero = 3;
  let numeroMayor = 0;

  datos
    .slice(1)
    .forEach(function(fila) {

      const id =
        String(
          fila[indiceId] || ""
        ).trim();

      const coincidencia =
        id.match(
          /^([A-Za-z]+)([-_]?)(\d+)$/
        );

      if (!coincidencia) {
        return;
      }

      const numero =
        Number(coincidencia[3]);

      if (numero > numeroMayor) {

        numeroMayor = numero;
        prefijo = coincidencia[1];
        separador = coincidencia[2];
        longitudNumero =
          coincidencia[3].length;

      }

    });

  return (
    prefijo +
    separador +
    String(numeroMayor + 1)
      .padStart(
        longitudNumero,
        "0"
      )
  );

}


function actualizarContenedorPredeterminadoChofer(
  hoja,
  idChofer,
  idUnidad
) {

  const datos =
    hoja
      .getDataRange()
      .getDisplayValues();

  const encabezados =
    datos[0].map(function(valor) {
      return String(valor || "").trim();
    });

  const indiceId =
    encabezados.indexOf("ID_Chofer");

  const indiceContenedor =
    encabezados.indexOf(
      "Contenedor_porDefecto"
    );

  if (
    indiceId === -1 ||
    indiceContenedor === -1
  ) {

    throw new Error(
      "Faltan las columnas ID_Chofer o Contenedor_porDefecto en Choferes."
    );

  }

  for (
    let i = 1;
    i < datos.length;
    i++
  ) {

    if (
      String(
        datos[i][indiceId] || ""
      ).trim() === idChofer
    ) {

      hoja
        .getRange(
          i + 1,
          indiceContenedor + 1
        )
        .setValue(idUnidad);

      return;

    }

  }

  throw new Error(
    "No se encontró el chofer seleccionado."
  );

}

function validarPrecinto(data){

  const noPrecinto = String(data.no_precinto || "").trim();

  if(!noPrecinto){
    return respuesta({
      ok:false,
      mensaje:"Debe indicar un número de precinto."
    });
  }

  const hoja = getSheet(SHEETS.PRECINTOS);

  if(!hoja){
    return respuesta({
      ok:false,
      mensaje:"No existe la hoja Precintos."
    });
  }

  const datos = hoja.getDataRange().getDisplayValues();

  datos.shift();

  const encontrado = datos.find(fila => {
    return String(fila[0]).trim() === noPrecinto;
  });

  if(!encontrado){
    return respuesta({
      ok:false,
      mensaje:"El precinto " + noPrecinto + " no existe en la base de datos."
    });
  }

  const estado = String(encontrado[2]).trim().toLowerCase();

  if(estado !== "disponible"){
    return respuesta({
      ok:false,
      mensaje:"El precinto existe, pero no está disponible. Estado actual: " + encontrado[2]
    });
  }

  return respuesta({
    ok:true,
    mensaje:"Precinto disponible.",
    data:{
      no_precinto: encontrado[0],
      estado: encontrado[2]
    }
  });

}

function obtenerInspeccionConduce(data) {

  const idConduce =
    String(
      data && data.idConduce
        ? data.idConduce
        : ""
    ).trim();

  if (!idConduce) {

    return respuesta({
      ok: false,
      mensaje:
        "Debe indicar el ID del conduce."
    });

  }

  const hoja =
    getSheet(
      SHEETS.INSPECCION_CONTENEDORES
    );

  if (!hoja) {

    return respuesta({
      ok: false,
      mensaje:
        "No existe la hoja Inspeccion_Contenedores."
    });

  }

  const datos =
    hoja
      .getDataRange()
      .getDisplayValues();

  if (datos.length <= 1) {

    return respuesta({
      ok: true,
      data: {
        encontrada: false,
        completada: false,
        numeroFormulario: "",
        estado: ""
      }
    });

  }

  const encabezados =
    datos[0].map(function(valor) {
      return String(valor || "").trim();
    });

  const indiceIdConduce =
    encabezados.indexOf("ID_Conduce");

  const indiceEstado =
    encabezados.indexOf("Estado");

  if (
    indiceIdConduce === -1 ||
    indiceEstado === -1
  ) {

    return respuesta({
      ok: false,
      mensaje:
        "Faltan las columnas ID_Conduce o Estado en Inspeccion_Contenedores."
    });

  }

  /*
   * El sistema acepta cualquiera de estos nombres
   * para la columna que contiene la numeración.
   */
  const nombresNumero = [
    "No_Formulario",
    "Numero_Formulario",
    "No_Inspeccion",
    "Numero_Inspeccion",
    "ID_Inspeccion",
    "ID_Formulario"
  ];

  let indiceNumero = -1;

  for (
    let i = 0;
    i < nombresNumero.length;
    i++
  ) {

    const indice =
      encabezados.indexOf(
        nombresNumero[i]
      );

    if (indice !== -1) {
      indiceNumero = indice;
      break;
    }

  }

  if (indiceNumero === -1) {

    return respuesta({
      ok: false,
      mensaje:
        "No se encontró la columna con el número del formulario de inspección."
    });

  }

  const coincidencias =
    datos
      .slice(1)
      .filter(function(fila) {

        return (
          String(
            fila[indiceIdConduce] || ""
          ).trim() === idConduce
        );

      });

  if (coincidencias.length === 0) {

    return respuesta({
      ok: true,
      data: {
        encontrada: false,
        completada: false,
        numeroFormulario: "",
        estado: ""
      }
    });

  }

  /*
   * Preferimos una inspección completada.
   * Si todavía no está completada, devolvemos
   * el estado actual para mostrarlo al supervisor.
   */
  let filaSeleccionada =
    coincidencias.find(function(fila) {

      const estado =
        String(
          fila[indiceEstado] || ""
        )
          .trim()
          .toLowerCase();

      return (
        estado === "completada" ||
        estado === "completado"
      );

    });

  if (!filaSeleccionada) {

    filaSeleccionada =
      coincidencias[
        coincidencias.length - 1
      ];

  }

  const estado =
    String(
      filaSeleccionada[indiceEstado] || ""
    ).trim();

  const estadoNormalizado =
    estado.toLowerCase();

  const completada =
    estadoNormalizado === "completada" ||
    estadoNormalizado === "completado";

  const numeroFormulario =
    String(
      filaSeleccionada[indiceNumero] || ""
    ).trim();

  return respuesta({
    ok: true,
    data: {
      encontrada: true,
      completada:
        completada &&
        numeroFormulario !== "",
      numeroFormulario:
        numeroFormulario,
      estado:
        estado
    }
  });

}

/** Defensa central: la visibilidad del frontend nunca sustituye el permiso backend. */
function validarAccionSoloLecturaPorRol_(accion, data) {
  var rol = "";
  try {
    rol = galUsuario_(data).rol;
  } catch (ignorar) {
    return "";
  }

  var gerencia = rol === "GERENCIA" || rol === "GERENTE";
  var supervisor = rol === "SUPERVISOR";
  var analista = rol === "ANALISTA";
  var decisionesDecomiso = ["aprobarDecomisoAlmacen", "rechazarDecomisoAlmacen"];
  var accionesReporteConsulta = ["generarArchivoReportesKPI", "enviarArchivoReportesKPI"];

  if (gerencia && decisionesDecomiso.indexOf(accion) < 0 && accionesReporteConsulta.indexOf(accion) < 0) {
    var prefijosMutacion = /^(guardar|crear|iniciar|finalizar|completar|subir|eliminar|registrar|recalcular|sincronizar|reconstruir|agregar|cambiar|aplicar|entregar|ajustar|actualizar|definir|asignar|autorizar|rechazar|aprobar)/;
    var enviosOperativos = [
      "enviarDocumentacionDespacho", "enviarDecomisoAprobacion",
      "enviarSolicitudCompraEPP", "enviarListadoJornadaExtraordinaria"
    ];
    if (prefijosMutacion.test(accion) || enviosOperativos.indexOf(accion) >= 0) {
      return "Gerencia dispone únicamente de consulta; solo puede aprobar o rechazar decomisos.";
    }
  }

  if (supervisor) {
    var accionesAlmacenNoPermitidas = [
      "guardarCamaraAlmacen", "cambiarEstadoCamaraAlmacen", "guardarTramoAlmacen",
      "aplicarCapacidadPisoAlmacen", "guardarSolicitanteAlmacen",
      "guardarMaterialEspecialAlmacen", "entregarMaterialEspecialAlmacen",
      "registrarAjusteOcupacionAlmacen",
      "guardarDecomisoAlmacen", "validarFisicamenteDecomisoAlmacen",
      "enviarDecomisoAprobacion", "aprobarDecomisoAlmacen",
      "rechazarDecomisoAlmacen", "registrarSalidaSAPDecomiso",
      "enviarReporteDecomisos", "enviarReporteAlmacen"
    ];
    if (accionesAlmacenNoPermitidas.indexOf(accion) >= 0) {
      return "Supervisión dispone únicamente de consulta en Gestión del almacén.";
    }
  }

  if (analista) {
    var accionesConfiguracionAlmacenNoPermitidas = [
      "guardarCamaraAlmacen", "cambiarEstadoCamaraAlmacen", "guardarTramoAlmacen",
      "aplicarCapacidadPisoAlmacen", "guardarSolicitanteAlmacen",
      "registrarAjusteOcupacionAlmacen"
    ];
    if (accionesConfiguracionAlmacenNoPermitidas.indexOf(accion) >= 0) {
      return "El Analista puede consultar el almacén, movilizar materiales, gestionar materiales especiales y registrar decomisos; no puede modificar su configuración.";
    }
  }

  return "";
}
