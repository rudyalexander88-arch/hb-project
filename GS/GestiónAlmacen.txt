/**
 * Guarda el plano cartesiano completo mediante la acción existente.
 * X = Mapa_Columna; Y = Mapa_Fila. El origen corresponde a la esquina
 * inferior izquierda y ninguna cámara puede superponerse con otra.
 */
function guardarDistribucionMapaAlmacen_(datos){
  return galRespuesta_(function(){
    galPermiso_(datos,'ADMIN');
    if(!datos.distribucion.length)throw Error('No se recibieron cámaras para guardar.');

    var bloqueo=LockService.getScriptLock();
    bloqueo.waitLock(30000);

    try{
      var hoja=getSheet(GAL_VERSIONES_HOJAS_BON_20260824.CAMARAS);
      if(!hoja)throw Error('No existe la hoja Config_Camaras.');

      var valores=hoja.getDataRange().getValues();
      if(valores.length<2)throw Error('No existen cámaras registradas.');

      var encabezados=valores[0].map(String);
      var columnas={
        id:encabezados.indexOf('ID_Camara'),
        fila:encabezados.indexOf('Mapa_Fila'),
        columna:encabezados.indexOf('Mapa_Columna'),
        ancho:encabezados.indexOf('Mapa_Ancho'),
        alto:encabezados.indexOf('Mapa_Alto'),
        modificacion:encabezados.indexOf('Fecha_Modificacion')
      };

      ['id','fila','columna','ancho','alto'].forEach(function(clave){
        if(columnas[clave]<0)throw Error('Falta una columna del mapa en Config_Camaras: '+clave+'.');
      });

      var existentes={};
      for(var f=1;f<valores.length;f++){
        var idExistente=galTexto_(valores[f][columnas.id]);
        if(idExistente)existentes[idExistente]=f;
      }

      var vistos={};
      var distribucion=datos.distribucion.map(function(camara){
        var id=galTexto_(camara.idCamara||camara.ID_Camara);
        if(!id||existentes[id]===undefined)throw Error('La cámara indicada no existe: '+(id||'sin identificación')+'.');
        if(vistos[id])throw Error('La cámara aparece más de una vez en la distribución: '+id+'.');
        vistos[id]=true;

        function numero(valor,nombre,minimo){
          var resultado=Number(valor);
          if(!isFinite(resultado)||Math.floor(resultado)!==resultado||resultado<minimo){
            throw Error('Valor no válido de '+nombre+' para la cámara '+id+'.');
          }
          return resultado;
        }

        return{
          id:id,
          fila:numero(camara.fila!==undefined?camara.fila:camara.Mapa_Fila,'coordenada Y',0),
          columna:numero(camara.columna!==undefined?camara.columna:camara.Mapa_Columna,'coordenada X',0),
          ancho:numero(camara.ancho!==undefined?camara.ancho:camara.Mapa_Ancho,'ancho',1),
          alto:numero(camara.alto!==undefined?camara.alto:camara.Mapa_Alto,'altura',1)
        };
      });

      for(var a=0;a<distribucion.length;a++){
        for(var b=a+1;b<distribucion.length;b++){
          var primera=distribucion[a],segunda=distribucion[b];
          if(primera.columna<segunda.columna+segunda.ancho&&
             primera.columna+primera.ancho>segunda.columna&&
             primera.fila<segunda.fila+segunda.alto&&
             primera.fila+primera.alto>segunda.fila){
            throw Error('Las cámaras '+primera.id+' y '+segunda.id+' se superponen.');
          }
        }
      }

      var indices=[columnas.fila,columnas.columna,columnas.ancho,columnas.alto].sort(function(a,b){return a-b});
      var primeraColumna=indices[0],ultimaColumna=indices[indices.length-1];

      if(ultimaColumna-primeraColumna===3){
        var rango=hoja.getRange(2,primeraColumna+1,valores.length-1,4);
        var bloque=rango.getValues();
        distribucion.forEach(function(camara){
          var fila=existentes[camara.id]-1;
          bloque[fila][columnas.fila-primeraColumna]=camara.fila;
          bloque[fila][columnas.columna-primeraColumna]=camara.columna;
          bloque[fila][columnas.ancho-primeraColumna]=camara.ancho;
          bloque[fila][columnas.alto-primeraColumna]=camara.alto;
        });
        rango.setValues(bloque);
      }else{
        ['fila','columna','ancho','alto'].forEach(function(clave){
          var rangoColumna=hoja.getRange(2,columnas[clave]+1,valores.length-1,1);
          var contenido=rangoColumna.getValues();
          distribucion.forEach(function(camara){contenido[existentes[camara.id]-1][0]=camara[clave]});
          rangoColumna.setValues(contenido);
        });
      }

      if(columnas.modificacion>=0){
        var fecha=new Date();
        var celdas=distribucion.map(function(camara){
          return hoja.getRange(existentes[camara.id]+1,columnas.modificacion+1).getA1Notation();
        });
        hoja.getRangeList(celdas).setValue(fecha);
      }

      return{cantidad:distribucion.length,mensaje:'Distribución del almacén guardada correctamente.'};
    }finally{
      try{bloqueo.releaseLock()}catch(ignorar){}
    }
  });
}
